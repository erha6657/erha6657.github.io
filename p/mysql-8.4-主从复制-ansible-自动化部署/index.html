MySQL 8.4 主从复制 Ansible 自动化部署

2026年5月19日星期二 阅读时长: 2 分钟

本次使用的架构是1主1从，采用的数据库是 MySQL 8.4 社区版。在搭建开始前已经进行好安装和初始用户注册，用户 root 密码为 Chen123456!，已经提前配置好 Ansible。

  • 主库: 192.168.10.51 (主)
  • 从库: 192.168.10.52 (从)

- name: 使用 Jinja2 模板修改 my.cnf 配置
  blockinfile:
    path: /etc/my.cnf
    insertafter: '^\[mysqld\]'
    block: "{{ lookup('template', 'my_cnf_block.j2') }}"
  notify: Restart MySQL



# -------- 主库 --------
- name: 主库 - 创建复制用户并授权
  shell: |
    mysql -uroot -p'{{ mysql_root_password }}' -e "
    CREATE USER IF NOT EXISTS '{{ repl_user }}'@'192.168.10.%' IDENTIFIED WITH mysql_native_password BY '{{ repl_password }}';
    GRANT REPLICATION SLAVE ON *.* TO '{{ repl_user }}'@'192.168.10.%';
    FLUSH PRIVILEGES;"
  when: "'master' in group_names"

# -------- 从库 --------
- name: 从库 - 检查当前主从状态
  shell: mysql -uroot -p'{{ mysql_root_password }}' -e "SHOW REPLICA STATUS\G" | grep "Replica_IO_Running.*Yes"
  register: replica_status
  ignore_errors: true
  changed_when: false
  when: "'slave' in group_names"
        
    - name: 启动从库 MySQL
      service:
        name: mysqld
        state: started

    - name: 配置主从关系并启动 REPLICA
      shell: |
        mysql -uroot -p'{{ mysql_root_password }}' -e "
        CHANGE REPLICATION SOURCE TO 
            SOURCE_HOST = '{{ master_ip }}',
            SOURCE_USER = '{{ repl_user }}',
            SOURCE_PASSWORD = '{{ repl_password }}',
            SOURCE_PORT = 3306,
            SOURCE_AUTO_POSITION = 1;
        START REPLICA;"
