在 Kubernetes 中，手动维护 Prometheus 的配置文件（ConfigMap）和自建服务发现非常繁琐。

Prometheus Operator 的核心思想是引入 Custom Resource Definitions (CRDs)。你不再需要直接修改 Prometheus 配置文件，而是通过声明 Kubernetes 资源来管理监控：

  • Prometheus: 定义一个 Prometheus 实例。
  • ServiceMonitor / PodMonitor: 声明式地定义谁需要被监控（通过 Label 筛选 Service）。
  • PrometheusRule: 定义告警规则。
  • Alertmanager: 定义告警通知实例。

安装 Prometheus Operator

我们将使用官方提供的 kube-prometheus 项目中的清单文件进行快速部署。

官网链接：https://prometheus-operator.dev/docs/getting-started/installation/

你可以从 GitHub 克隆 kube-prometheus：

apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: mysql-up-alerts
  namespace: monitoring
  labels:
    release: prometheus
    role: alert-rules
spec:
  groups:
  - name: mysql.rules
    rules:
    - alert: MySQLDown
      expr: mysql_up == 0
      for: 1m
      labels:
        severity: critical
      annotations:
        summary: "MySQL 实例挂了！ (实例: {{ $labels.instance }})"
        description: "MySQL 服务 {{ $labels.job }} 在命名空间 {{ $labels.namespace }} 中已停止运行或无法连接。"

apiVersion: v1
kind: Secret
metadata:
  name: alertmanager-main
  namespace: monitoring
type: Opaque
stringData:
  alertmanager.yaml: |
    # =============================================================================
    # Alertmanager 全局配置
    # =============================================================================
    global:
      # SMTP 服务器地址和端口
      # 格式: '主机名:端口'
      # 465端口是SSL加密端口，587端口是STARTTLS端口
      smtp_smarthost: 'smtp.163.com:465'
      
      # 发件人邮箱地址
      # 必须与smtp_auth_username一致或属于同一账户
      smtp_from: 'chenyangqit@163.com'
      
      # SMTP认证用户名（通常是完整的邮箱地址）
      smtp_auth_username: 'chenyangqit@163.com'
      
      # SMTP认证密码或授权码
      # 对于163邮箱，需要使用授权码而不是登录密码
      # 授权码可以在邮箱设置中生成
      smtp_auth_password: 'PTsexxxxx95panw'
      
      # 是否要求TLS加密
      # 使用465端口时设为false（因为465端口使用隐式SSL）
      # 使用587端口时设为true（因为587端口使用显式STARTTLS）
      smtp_require_tls: false

    # =============================================================================
    # 告警路由配置
    # =============================================================================
    route:
      # 告警分组依据的标签列表
      # 具有相同这些标签值的告警会被分到同一组
      group_by: ['alertname', 'namespace']
      
      # 初次等待时间：当第一个告警到达后，等待多长时间才发送通知
      # 这样可以将同一时间段的多个相关告警合并发送
      group_wait: 10s
      
      # 组间隔时间：在发送第一个通知后，等待多长时间才发送更新的通知
      # 用于控制告警更新的频率
      group_interval: 10s
      
      # 重复间隔：如果告警持续触发，重复发送通知的时间间隔
      # 避免对持续存在的问题发送过多通知
      repeat_interval: 1h
      
      # 默认接收器：当告警不匹配任何子路由时使用的接收器
      receiver: 'email-163'
      
      # 子路由规则列表（按顺序匹配）
      routes:
        # 第一条子路由：处理严重( critical )级别的告警
        - matchers:
            # 匹配条件：severity标签值为"critical"
            # 语法说明：
            # =   : 精确等于
            # !=  : 不等于  
            # =~  : 正则匹配
            # !~  : 正则不匹配
            - severity="critical"
          
          # 使用哪个接收器发送通知
          receiver: 'email-163'
          
          # 严重告警的组间隔更短（30秒），响应更快
          group_interval: 30s
          
          # 严重告警的重复间隔更短（10分钟），提醒更频繁
          repeat_interval: 10m
          
          # 严重告警的分组方式：只按告警名称分组
          # 这样每个严重告警都会单独发送，不会被其他标签分组
          group_by: ['alertname', 'namespace']
          
          # 是否继续匹配后续路由
          # false: 匹配到此路由后停止继续匹配
          # true:  匹配到此路由后继续匹配其他路由
          continue: false

    # =============================================================================
    # 模板文件配置
    # =============================================================================
    templates:
      # 自定义模板文件路径
      # 支持通配符，如: "/etc/alertmanager/template/*.tmpl"
      # 模板文件用于定义邮件主题、HTML内容、文本内容等的格式
      - "/etc/alertmanager/template/email-html.tmpl"

    # =============================================================================
    # 接收器配置
    # =============================================================================
    receivers:
      # 接收器名称，在路由中引用
      - name: 'email-163'
        
        # 邮件配置列表（可以配置多个邮件接收配置）
        email_configs:
        - to: "chenyangqit@163.com"
          
          # 是否在告警恢复时发送解决通知
          send_resolved: true
          
          # 邮件头配置
          headers:
            # 邮件主题模板
            # 使用模板语法动态生成主题
            # {{ template "email.custom.subject" . }} 调用自定义主题模板
            # . 表示传递给模板的上下文数据
            Subject: '{{ template "email.custom.subject" . }}'
          
          # HTML格式的邮件内容模板
          # 用于支持HTML的邮件客户端显示富文本内容
          html: '{{ template "email.custom.html" . }}'
          
          # 纯文本格式的邮件内容模板  
          # 用于不支持HTML的邮件客户端或作为备选显示
          # 提供更好的兼容性
          text: '{{ template "email.custom.text" . }}'

    # =============================================================================
    # 告警抑制规则（Inhibit Rules）
    # 核心作用：当高级别/特定告警触发时，抑制低级别/关联告警的发送，减少冗余通知
    # 执行逻辑：源告警（source）满足条件时，抑制目标告警（target）的发送
    # 匹配规则：只有源告警和目标告警在 "equal" 字段的标签上完全一致，规则才生效
    # =============================================================================
    inhibit_rules:
      # 规则1：Critical级别告警抑制同类型的Warning/Info级别告警
      # 场景：如MySQLDown(critical)触发时，不再发送MySQLDown(warning/info)告警
      - equal:
          # 源/目标告警必须同命名空间（如default）
          - "namespace"
          # 源/目标告警必须同名（如MySQLDown）
          - "alertname"
        # 源告警匹配条件（触发抑制的告警）
        source_matchers:
          # 源告警是critical级别
          - "severity = critical"
        # 目标告警匹配条件（被抑制的告警）
        target_matchers:
          # 目标告警是warning或info级别
          - "severity =~ warning|info"

      # 规则2：Warning级别告警抑制同类型的Info级别告警
      # 场景：如MySQLHighCPU(warning)触发时，不再发送MySQLHighCPU(info)告警
      - equal:
          # 源/目标告警必须同命名空间
          - "namespace"
          # 源/目标告警必须同名
          - "alertname"
        # 源告警匹配条件
        source_matchers:
          # 源告警是warning级别
          - "severity = warning"
        # 目标告警匹配条件
        target_matchers:
          # 目标告警是info级别
          - "severity = info"

      # 规则3：InfoInhibitor告警抑制同命名空间下所有Info级别告警
      # 场景：手动触发InfoInhibitor告警时，屏蔽该命名空间所有info级告警（如维护期间）
      - equal:
          # 仅要求同命名空间（不要求告警名相同）
          - "namespace"
        # 源告警匹配条件
        source_matchers:
          # 源告警是自定义的"抑制开关"告警
          - "alertname = InfoInhibitor"
        # 目标告警匹配条件
        target_matchers:
          # 目标告警是info级别
          - "severity = info"

    # =============================================================================
    # 配置说明总结
    # =============================================================================

metadata:
  labels:
    app.kubernetes.io/component: alert-router
    app.kubernetes.io/instance: main
    app.kubernetes.io/name: alertmanager
    app.kubernetes.io/part-of: kube-prometheus
    app.kubernetes.io/version: 0.28.1
  name: main
  namespace: monitoring
spec:
  image: quay.io/prometheus/alertmanager:v0.28.1
  nodeSelector:
    kubernetes.io/os: linux
  podMetadata:
    labels:
      app.kubernetes.io/component: alert-router
      app.kubernetes.io/instance: main
      app.kubernetes.io/name: alertmanager
      app.kubernetes.io/part-of: kube-prometheus
  replicas: 3
  resources:
    limits:
      cpu: 100m
      memory: 100Mi
    requests:
      cpu: 4m
      memory: 100Mi
  secrets: []
  securityContext:
    fsGroup: 2000
    runAsNonRoot: true
    runAsUser: 1000
  serviceAccountName: alertmanager-main
  version: 0.28.1
  configSecret: alertmanager-main
  volumes:
    - name: alertmanager-templates
      secret:
        secretName: alertmanager-main
        items:
          - key: email-html.tmpl
            path: email-html.tmpl
  volumeMounts:
    - name: alertmanager-templates
