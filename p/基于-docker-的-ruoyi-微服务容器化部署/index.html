version: '3.8'

networks:
  ruoyi-net:
    name: ruoyi-net
    driver: bridge

services:
  mysql:
    image: swr.cn-north-4.myhuaweicloud.com/ddn-k8s/docker.io/mysql:5.7
    container_name: mysql
    ports:
      - "3306:3306"
    environment:
      MYSQL_ROOT_PASSWORD: "Chen123456!"
      MYSQL_DATABASE: "ruoyi"
      TZ: Asia/Shanghai
    volumes:
      - ./mysql/conf/my.cnf:/etc/mysql/conf.d/my.cnf
      - ./mysql/init:/docker-entrypoint-initdb.d
      - ./mysql/data:/var/lib/mysql
    networks:
      - ruoyi-net

  redis:
    image: swr.cn-north-4.myhuaweicloud.com/ddn-k8s/docker.io/redis:7.2-alpine
    container_name: redis
    ports:
      - "6379:6379"
    command: redis-server --requirepass "Chen123456!" --appendonly yes
    networks:
      - ruoyi-net

  minio:
    image: swr.cn-north-4.myhuaweicloud.com/ddn-k8s/docker.io/minio/minio:latest
    container_name: minio
    ports:
      - "9000:9000"
      - "9001:9001"
    environment:
      MINIO_ROOT_USER: "admin"
      MINIO_ROOT_PASSWORD: "Chen123456!"
    volumes:
      - ./minio/data:/data
    command: server /data --console-address ":9001"
    networks:
      - ruoyi-net

  nacos:
    image: swr.cn-north-4.myhuaweicloud.com/ddn-k8s/docker.io/nacos/nacos-server:v2.5.2
    container_name: nacos
    ports:
      - "8848:8848"
    environment:
      MODE: "standalone"
      SPRING_DATASOURCE_PLATFORM: "mysql"
      MYSQL_SERVICE_HOST: "mysql"
      MYSQL_SERVICE_PORT: "3306"
      MYSQL_SERVICE_DB_NAME: "ruoyi"
      MYSQL_SERVICE_USER: "root"
      MYSQL_SERVICE_PASSWORD: "Chen123456!"
      JVM_XMS: "256m"
      JVM_XMX: "256m"
      TZ: Asia/Shanghai
    depends_on:
      - mysql
    networks:
      - ruoyi-net

  sentinel:

cd /root/code/ruoyi/sql

# 3. 一键替换 SQL 中的所有旧 IP 为 Docker 容器名！
sed -i -E 's/(192\.168\.[0-9]+\.[0-9]+|127\.0\.0\.1):3306/mysql:3306/g' ry_config_*.sql
sed -i -E 's/(192\.168\.[0-9]+\.[0-9]+|127\.0\.0\.1):6379/redis:6379/g' ry_config_*.sql
sed -i -E 's/(192\.168\.[0-9]+\.[0-9]+|127\.0\.0\.1):9000/minio:9000/g' ry_config_*.sql


cd /root/code/ruoyi/sql

# 1. 替换 MySQL 库名和地址 (把 localhost 换成 mysql，ry-cloud 换成 ruoyi)
sed -i 's/localhost:3306\/ry-cloud/mysql:3306\/ruoyi/g' ry_config_*.sql

# 2. 替换 MySQL 密码 (把原版的 password: password 换成你的密码)
sed -i 's/password: password/password: Chen123456!/g' ry_config_*.sql

# 3. 替换 Redis 地址 (把 host: localhost 换成 host: redis)
sed -i 's/host: localhost/host: redis/g' ry_config_*.sql

# 4. 替换 Redis 密码 (把空密码替换成你的密码)
# 注意：原版配置中 password: 后面是空的，所以我们要精确匹配并替换
sed -i 's/port: 6379\r\n    password:/port: 6379\r\n    password: Chen123456!/g' ry_config_*.sql
sed -i 's/port: 6379\n    password:/port: 6379\n    password: Chen123456!/g' ry_config_*.sql
